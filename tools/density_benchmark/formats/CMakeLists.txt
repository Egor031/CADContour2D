cmake_minimum_required(VERSION 3.24)
project(DensityFormatsExperiment LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The experiment requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(density_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core)
find_package(ZLIB REQUIRED)
find_package(TIFF REQUIRED)
find_package(GTest CONFIG REQUIRED)
include(CTest)
include(GoogleTest)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${density_root}"
    OUTPUT_VARIABLE density_head OUTPUT_STRIP_TRAILING_WHITESPACE)
add_library(density_harness STATIC ../dataset.cpp ../measurement.cpp)
target_include_directories(density_harness PUBLIC .. "${density_root}/src")
target_link_libraries(density_harness PUBLIC Qt6::Core PRIVATE Psapi)
target_compile_definitions(density_harness PRIVATE
    DENSITY_BUILD_HEAD="${density_head}" DENSITY_BUILD_CONFIG="$<CONFIG>"
    DENSITY_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
add_library(density_formats STATIC count_container.cpp tiff_adapter.cpp format_support.cpp)
target_include_directories(density_formats PUBLIC .)
target_link_libraries(density_formats PUBLIC density_harness ZLIB::ZLIB TIFF::TIFF)
add_executable(density_formats_benchmark benchmark.cpp)
target_link_libraries(density_formats_benchmark PRIVATE density_formats)
target_compile_definitions(density_formats_benchmark PRIVATE DENSITY_SOURCE_DIR="${density_root}")
add_executable(density_formats_tests tests.cpp)
target_link_libraries(density_formats_tests PRIVATE density_formats GTest::gtest_main)
target_compile_definitions(density_formats_tests PRIVATE
    DENSITY_FORMATS_EXE="$<TARGET_FILE:density_formats_benchmark>")
add_dependencies(density_formats_tests density_formats_benchmark)
foreach(target density_harness density_formats density_formats_benchmark density_formats_tests)
    target_compile_options(${target} PRIVATE /W4 /permissive- /fp:precise /utf-8)
endforeach()
gtest_discover_tests(density_formats_tests DISCOVERY_MODE PRE_TEST
    PROPERTIES LABELS density_formats TIMEOUT 90)
